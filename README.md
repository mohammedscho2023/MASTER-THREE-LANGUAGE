function oromoRespell(text, voice){
  let t = String(text).toLowerCase().replace(/[’‘`´]/g, "'");
  t = t.replace(/(\p{L})'(\p{L})/gu, '$1$2').replace(/'/g, '');
  if (voice === 'sw'){
    return t.replace(/ph/g,'p').replace(/dh/g,'d').replace(/x/g,'t').replace(/q/g,'k').replace(/c(?!h)/g,'ch');
  }
  if (voice === 'tr'){
    return t.replace(/ch/g,'ç').replace(/sh/g,'ş').replace(/c/g,'ç').replace(/j/g,'c').replace(/ph/g,'p').replace(/dh/g,'d').replace(/x/g,'t').replace(/q/g,'k').replace(/w/g,'v');
  }
  return t;
}

      if (!letterIdx.length) return esc(token);
      const start = letterIdx[0], end = letterIdx[letterIdx.length-1] + 1;
      const word = chars.slice(start, end);
      const boldLen = Math.max(1, Math.ceil(word.length * 0.4));
      return esc(chars.slice(0,start).join('')) + '<b>' + esc(word.slice(0,boldLen).join('')) + '</b>' + esc(word.slice(boldLen).join('')) + esc(chars.slice(end).join(''));

/* Adaptive pick: prefer words that are new or weak (Leitner-style boxes) */
function pickVocab(){
  const tgt = tgtLang();
  const weights = VOCAB.map(v => {
    const keys = Object.keys(store.data.progress).filter(k => k.includes('-' + v.id + '-') || k.endsWith('-' + v.id));
    if (!keys.length) return 3;
    const box = Math.min(...keys.map(k => store.data.progress[k].box));
    return Math.max(0.4, 3 - box * 0.4);
  });
  const total = weights.reduce((a,b) => a+b, 0);
  let r = Math.random() * total;
  for (let i = 0; i < VOCAB.length; i++){ r -= weights[i]; if (r <= 0) return VOCAB[i]; }
  return rand(VOCAB);
}

    prompt:'Translate into ' + tgtName() + ':', promptSub:'🎯', scene:sceneFor({ emoji:s.emoji }),
    sourceText:s[src], sourceLang:src, sourceClass:L(src).class,
    choices:shuffle([{ text:s[tgt], correct:true, lang:tgt }, ...wrongs.map(w => ({ text:w[tgt], correct:false, lang:tgt }))]),
    correct:s[tgt], item:{ id:s.id, emoji:s.emoji, en:s.en }, explain:'"' + s[src] + '" in ' + tgtName() + ' is "' + s[tgt] + '".',
    speakCorrect:s[tgt], speakLang:speechFor(tgt),
    narration:[{ text:s[src], lang:speechFor(src) }] };
}
function buildSpeakingDrill(){
  const item = pickVocab(), tgt = tgtLang(), target = wordFor(item, tgt);
  return { kind:'speakingdrill', id:'spk-' + item.id + '-' + tgt, skill:'Speaking Drill · ' + tgtName(), prompt:'Say this aloud in ' + tgtName() + ':',
    promptSub:'🗣️ Listen to the model, speak clearly, then rate yourself.', scene:sceneFor(item), word:target, item,
    explain:'Honest self-assessment is a real skill. Noticing what was hard tells you what to practise.',
    speakCorrect:target, speakLang:speechFor(tgt), narration:[{ text:target, lang:speechFor(tgt) }] };
}
function buildWritingPrompt(){
  const item = pickVocab(), tgt = tgtLang(), w = wordFor(item, tgt);
  const scenarios = [
    'Write one sentence introducing yourself using "' + w + '" (' + item.en + ').',

    promptSub:'✍️ Write in the box, then rate yourself against the rubric.', scene:sceneFor(item), targetWord:w, item,

}

function buildTodaysPath(){
  const total = store.count(); let kinds;

/* One generic handler for single-choice rounds */
function answer(i, points){
  const g = S.game; if (g.picked !== null) return;
  const c = g.round.choices ? g.round.choices[i] : g.round.options[i];
  const text = typeof c === 'string' ? c : c.text;
  const ok = typeof c === 'string' ? c === g.round.correct : c.correct;
  g.picked = text; g.result = ok ? 'ok' : 'no';
  store.record(g.round.id, ok);
  if (ok){ S.sessionStats.correct++; store.addPoints(points); haptic('ok'); if (g.kind === 'reading') store.bumpSkill('reading'); if (g.round.speakLang && g.kind !== 'reading') Voice.say(g.round.speakCorrect, g.round.speakLang); }
  else { S.sessionStats.retried++; haptic('no'); }

/* Speech recognition */
const recog = {
  Recognition: window.SpeechRecognition || window.webkitSpeechRecognition, instance:null, listening:false,
  start(lang, onResult, onEnd){
    if (!this.Recognition) return false;
    try {
      const rc = this.instance = new this.Recognition();
      rc.lang = lang || 'en-US'; rc.continuous = false; rc.interimResults = false; rc.maxAlternatives = 3;
      rc.onresult = e => { const alts = []; for (let i = 0; i < e.results[0].length; i++) alts.push(e.results[0][i].transcript); onResult(alts); };
      rc.onerror = () => { this.listening = false; onEnd(true); };
      rc.onend = () => { this.listening = false; onEnd(false); };
      rc.start(); this.listening = true; return true;
    } catch(e){ return false; }
  },
  stop(){ try { if (this.instance) this.instance.stop(); } catch(e){} this.listening = false; }
};
function renderSpeechGame(g){
  const r = g.round, supported = !!recog.Recognition, tgt = tgtLang();
  let status = supported ? 'Tap the microphone and say the word.' : 'Speech recognition is not available in this browser. Say the word aloud, then tap "I said it".';
  if (recog.listening) status = 'Listening… say the word now.';
  if (g.speechMiss && !g.result) status = 'Did not catch that. Try again or tap "I said it".';
  return '<div style="display:grid;grid-template-columns:1fr 1fr;gap:10px">' + r.words.map(w => wordCardHtml(w.lang, w.text, w.translit, { highlight:w.lang === tgt })).join('') + '</div>' +

function toggleSpeech(){
  if (recog.listening){ recog.stop(); render(); return; }
  const g = S.game, r = g.round;
  const started = recog.start(r.speakLang, alts => {
    const norm = s => s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/[^\p{L}\p{N}]+/gu,' ').trim();
    const target = norm(r.word);
    const ok = alts.some(a => { const n = norm(a); return n === target || (n && target.includes(n) && n.length >= target.length * 0.7) || (n && n.includes(target)); });
    if (S.game !== g || g.result) return;
    if (ok){ g.result = 'ok'; store.record(r.id, true); S.sessionStats.correct++; store.addPoints(3); haptic('ok'); Voice.say(r.word, r.speakLang); }
    else { g.speechMiss = true; haptic('no'); }
    render();
  }, err => { if (S.screen === 'game' && S.game === g){ if (err) g.speechMiss = true; render(); } });
  if (started) render(); else toast('Could not start the microphone.');
}

}

function extAdvance(g){
  const ext = extState(g), turns = g.round.turns;
  ext.turn++; ext.wrongPicks = [];
  while (ext.turn < turns.length && turns[ext.turn].who === 'partner') ext.turn++;
  /* show partner lines that precede the next learner turn */
  if (ext.turn >= turns.length){ g.result = 'ok'; store.record(g.round.id, !ext.hadMiss); if (!ext.hadMiss) S.sessionStats.correct++; else S.sessionStats.retried++; }
}
function extPick(i){
  const g = S.game, ext = extState(g), cur = g.round.turns[ext.turn];
  if (!cur || cur.who !== 'you' || g.result) return;
  const text = cur.options[i];
  if (text === cur.correct){
    ext.log[ext.turn] = text; haptic('ok'); store.addPoints(3); Voice.say(text, speechFor(tgtLang()));
    extAdvance(g);
  } else { ext.wrongPicks.push(text); ext.hadMiss = true; haptic('no'); }
  render();
}

/* Keyboard: 1-4 to answer, Enter for next, Space to replay audio */
document.addEventListener('keydown', e => {
  if (S.screen !== 'game' || !S.game) return;
  const t = e.target && e.target.tagName;
  if (t === 'TEXTAREA' || t === 'INPUT') return;
  const g = S.game;
  if (g.result){ if (e.key === 'Enter'){ e.preventDefault(); nextActivity(); } return; }
  if (e.key >= '1' && e.key <= '4'){
    const i = parseInt(e.key, 10) - 1;
    if (g.kind === 'extended'){ const cur = g.round.turns[extState(g).turn]; if (cur && cur.who === 'you' && cur.options[i]) extPick(i); }
    else if ((g.round.choices && g.round.choices[i]) || (g.round.options && g.round.options[i])){ if (['bridge','script','numbers','dialogue','difference','reading','grammar','translation'].includes(g.kind)) pickAnswer(i); }
  } else if (e.key === ' ' && !['BUTTON'].includes(t)){ e.preventDefault(); replayVoice(); }
});
